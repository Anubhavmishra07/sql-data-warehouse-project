/*
===============================================================================
Load Silver: ERP Location Information
===============================================================================
Purpose:
    Clean and transform ERP location data from the Bronze layer before
    loading it into the Silver layer.

Transformations:
    1. Remove '-' from customer IDs.
    2. Remove unwanted whitespace/invisible characters from country values.
    3. Standardize country names.
    4. Convert NULL or invalid country values to 'Unknown'.
===============================================================================
*/

TRUNCATE TABLE silver_erp_loc_a101;

INSERT INTO silver_erp_loc_a101 (
    cid,
    cntry
)

WITH cleaned_data AS (

    SELECT
        -- Remove '-' from customer IDs
        REPLACE(cid, '-', '') AS cid,

        -- Remove whitespace and hidden characters from country values
        REGEXP_REPLACE(cntry, '[[:space:]]+', '') AS cleaned_cntry

    FROM bronze_erp_loc_a101
)

SELECT
    cid,

    -- Standardize country values
    CASE
        WHEN UPPER(cleaned_cntry) = 'DE'
            THEN 'Germany'

        WHEN UPPER(cleaned_cntry) IN ('US', 'USA')
            THEN 'United States'

        WHEN cleaned_cntry IS NULL
             OR cleaned_cntry = ''
            THEN 'Unknown'

        ELSE cleaned_cntry
    END AS cntry

FROM cleaned_data;
