/*
===============================================================================
Quality Checks: Bronze CRM Product Information
===============================================================================
Purpose:
    Validate the quality, consistency, and integrity of CRM product data
    before loading it into the Silver layer.
===============================================================================
*/

USE datawarehouse;


-- ============================================================================
-- Review Raw Product Data
-- ============================================================================

SELECT *
FROM bronze_crm_prd_info;


-- ============================================================================
-- Check 1: Identify NULL or Duplicate Product IDs
-- ============================================================================

SELECT
    prd_id,
    COUNT(*) AS record_count
FROM bronze_crm_prd_info
GROUP BY prd_id
HAVING COUNT(*) > 1
    OR prd_id IS NULL;


-- ============================================================================
-- Check 2: Validate Product Category Key
-- ============================================================================
-- The first 5 characters of prd_key represent the category ID.
-- Verify that every category ID exists in the ERP product category table.
-- ============================================================================

SELECT
	REPLACE(SUBSTRING(prd_key, 1, 5), '-', '_') AS cat_id
FROM bronze_crm_prd_info AS p
WHERE REPLACE(SUBSTRING(prd_key, 1, 5), '-', '_') NOT in (
    SELECT id 
    FROM bronze_erp_px_cat_g1v2 
);


-- ============================================================================
-- Check 3: Validate Product Key Consistency
-- ============================================================================
-- The product ID portion of prd_key should exist in the CRM sales details.
-- ============================================================================

SELECT
SUBSTRING(p.prd_key, 7) as prd_key
FROM bronze_crm_prd_info AS p
WHERE SUBSTRING(p.prd_key, 7) NOT in (
    SELECT sls_prd_key
    FROM bronze_crm_sales_details
);



-- ============================================================================
-- Check 4: Detect Leading or Trailing Spaces
-- ============================================================================

SELECT
    SUM(prd_key <> TRIM(prd_key)) AS prd_key_spaces,
    SUM(prd_nm <> TRIM(prd_nm)) AS prd_nm_spaces,
    SUM(prd_line <> TRIM(prd_line)) AS prd_line_spaces
FROM bronze_crm_prd_info;


-- ============================================================================
-- Check 5: Identify NULL or Negative Product Costs
-- ============================================================================

SELECT
    prd_cost
FROM bronze_crm_prd_info
WHERE prd_cost < 0
   OR prd_cost IS NULL;

-- ============================================================================
-- Check 6: Review Product Line Values
-- ============================================================================
-- Review distinct values to identify inconsistent or unexpected categories.
-- ============================================================================

SELECT DISTINCT
    prd_line
FROM bronze_crm_prd_info
ORDER BY prd_line;


-- ============================================================================
-- Check 7: Validate Product Date Ranges
-- ============================================================================
-- The product end date should not be earlier than the start date.
-- ============================================================================

SELECT
    prd_start_dt,
    prd_end_dt
FROM bronze_crm_prd_info
WHERE prd_end_dt < prd_start_dt;
