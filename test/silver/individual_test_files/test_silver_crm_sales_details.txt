/*
===============================================================================
Quality Checks: silver CRM Sales Details
===============================================================================
Purpose:
    Validate the quality, consistency, and integrity of CRM sales data
    before loading it into the Bronze layer.
===============================================================================
*/

USE datawarehouse;


-- ============================================================================
-- Check 1: Detect Leading or Trailing Spaces in Order Numbers
-- ============================================================================

SELECT *
FROM silver_crm_sales_details
WHERE sls_ord_num <> TRIM(sls_ord_num);


-- ============================================================================
-- Check 2: Validate Product Keys
-- ============================================================================
-- Every product key in the sales data should exist in the Silver product
-- dimension.
-- ============================================================================

SELECT
    s.sls_prd_key
FROM silver_crm_sales_details AS s
WHERE NOT EXISTS (
    SELECT 1
    FROM silver_crm_prd_info AS p
    WHERE p.prd_key = s.sls_prd_key
);


-- ============================================================================
-- Check 3: Validate Customer IDs
-- ============================================================================
-- Every customer ID in the sales data should exist in the Silver customer
-- dimension.
-- ============================================================================

SELECT
    s.sls_cust_id
FROM silver_crm_sales_details AS s
WHERE NOT EXISTS (
    SELECT 1
    FROM silver_crm_cust_info AS c
    WHERE c.cst_id = s.sls_cust_id
);


-- ============================================================================
-- Check 4: Validate Order Dates
-- ============================================================================
-- Order dates should be valid 8-digit YYYYMMDD values.
-- ============================================================================

SELECT
    sls_order_dt
FROM silver_crm_sales_details
WHERE sls_order_dt <= 0
   OR LENGTH(sls_order_dt) <> 10;


-- ============================================================================
-- Check 5: Validate Ship Dates
-- ============================================================================
-- Ship dates should be valid 8-digit YYYYMMDD values.
-- ============================================================================

SELECT
    sls_ship_dt
FROM silver_crm_sales_details
WHERE sls_ship_dt <= 0
   OR LENGTH(sls_ship_dt) <> 10;


-- ============================================================================
-- Check 6: Validate Due Dates
-- ============================================================================
-- Due dates should be valid 8-digit YYYYMMDD values.
-- ============================================================================

SELECT
    sls_due_dt
FROM silver_crm_sales_details
WHERE sls_due_dt <= 0
  OR LENGTH(sls_due_dt) <> 10;


-- ============================================================================
-- Check 7: Validate Sales Date Sequence
-- ============================================================================
-- The order date should not be later than the ship date or due date.
-- ============================================================================

SELECT
    sls_ord_num,
    sls_order_dt,
    sls_ship_dt,
    sls_due_dt
FROM silver_crm_sales_details
WHERE sls_order_dt > sls_ship_dt
   OR sls_order_dt > sls_due_dt;


-- ============================================================================
-- Check 8: Validate Sales, Quantity, and Price
-- ============================================================================
-- Sales should equal quantity × price.
-- Sales, quantity, and price should all contain valid positive values.
-- ============================================================================

SELECT DISTINCT
    sls_sales,
    sls_quantity,
    sls_price
FROM silver_crm_sales_details
WHERE sls_sales <> sls_quantity * sls_price
   OR sls_sales IS NULL
   OR sls_quantity IS NULL
   OR sls_price IS NULL
   OR sls_sales <= 0
   OR sls_quantity <= 0
   OR sls_price <= 0
ORDER BY
    sls_sales,
    sls_quantity,
    sls_price;
