/*
===============================================================================
Quality Checks: Gold Layer
===============================================================================
Purpose:
    Validate the final Gold-layer Customer Dimension, Product Dimension,
    and Sales Fact after combining and transforming data from the Silver layer.
===============================================================================
*/


/*
===============================================================================
CUSTOMER DIMENSION
===============================================================================
*/


-- ============================================================================
-- Check 1: Review Gold Customer Dimension
-- ============================================================================

SELECT *
FROM gold_dim_customers;


-- ============================================================================
-- Check 2: Review Gender Values
-- ============================================================================
-- Verify that gender values are standardized correctly in the Gold layer.
-- ============================================================================

SELECT DISTINCT
    gender
FROM gold_dim_customers
ORDER BY gender;


/*
===============================================================================
PRODUCT DIMENSION
===============================================================================
*/


-- ============================================================================
-- Check 3: Review Gold Product Dimension
-- ============================================================================

SELECT *
FROM gold_dim_products;


-- ============================================================================
-- Check 4: Check for Duplicate Product Numbers
-- ============================================================================
-- Each current product should have only one record in the Gold dimension.
-- An empty result indicates that there are no duplicate product numbers.
-- ============================================================================

SELECT
    product_number,
    COUNT(*) AS record_count

FROM gold_dim_products

GROUP BY
    product_number

HAVING COUNT(*) > 1;


-- ============================================================================
-- Check 5: Review Product Line Values
-- ============================================================================
-- Verify that product line values are standardized correctly.
-- ============================================================================

SELECT DISTINCT
    product_line
FROM gold_dim_products
ORDER BY product_line;


/*
===============================================================================
SALES FACT
===============================================================================
*/


-- ============================================================================
-- Check 6: Review Gold Sales Fact
-- ============================================================================

SELECT *
FROM gold_fact_sale;


-- ============================================================================
-- Check 7: Validate Dimension Relationships
-- ============================================================================
-- Every sales fact should have a matching Customer and Product dimension
-- record through its respective surrogate keys.
--
-- If this query returns any rows, the corresponding dimension key is missing.
-- ============================================================================

SELECT
    f.order_number,
    f.product_key,
    f.customer_key

FROM gold_fact_sale AS f

LEFT JOIN gold_dim_customers AS c
    ON c.customer_key = f.customer_key

LEFT JOIN gold_dim_products AS p
    ON p.product_key = f.product_key

WHERE c.customer_key IS NULL
   OR p.product_key IS NULL;
